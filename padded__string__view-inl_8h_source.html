47inline bool padded_string_view::has_sufficient_padding() const noexcept {
48 if (padding() >= SIMDJSON_PADDING) {
49 return true;
50 }
51 size_t missing_padding = SIMDJSON_PADDING - padding();
52 if(length() < missing_padding) { return false; }
53
54 for (size_t i = length() - missing_padding; i < length(); i++) {
55 char c = data()[i];
56 if (c != ' ' && c != '\t' && c != '\n' && c != '\r') {
57 return false;
58 }
59 }
60 return true;
61}

81inline padded_string_view pad(std::string& s) noexcept {
82 size_t existing_padding = 0;
83 for (size_t i = s.size(); i > 0; i--) {
84 char c = s[i - 1];
85 if (c == ' ' || c == '\t' || c == '\n' || c == '\r') {
86 existing_padding++;
87 } else {
88 break;
89 }
90 }
91 size_t needed_padding = 0;
92 if (existing_padding < SIMDJSON_PADDING) {
93 needed_padding = SIMDJSON_PADDING - existing_padding;
94 s.append(needed_padding, ' ');
95 }
96
97 return padded_string_view(s.data(), s.size() - needed_padding, s.size());
98}

151inline padded_input::padded_input(const std::string &s)
152 : storage(simdjson::padded_string_view{}) {
153 const size_t len = s.size();
154 const size_t cap = s.capacity();
155 // Here we have the string content from data() to data() + size(),
156 // but the memory is accessible from data() to data() + capacity().
157 const size_t needed_padding = (cap - len) < simdjson::SIMDJSON_PADDING
158 ? simdjson::SIMDJSON_PADDING - (cap - len) : 0;
159 if (needed_padding > 0 && needs_allocation(s.data(), cap, needed_padding)) {
160 storage = simdjson::padded_string(s);
161 } else {
162 storage = simdjson::padded_string_view(
163 s.data(), len, len + simdjson::SIMDJSON_PADDING);
164 }
165}

185inline padded_string_view slice_at(padded_string_view data, char delimiter,
186 size_t block_size, size_t index) noexcept {
187 if (block_size == 0 || index > data.size() / block_size) { return {}; }
188 const size_t raw_begin = index * block_size;
189 if (raw_begin >= data.size()) { return {}; }
190
191 auto snap = [&](size_t want) -> size_t {
192 if (want >= data.size()) { return data.size(); }
193 const void *p = std::memchr(data.data() + want, delimiter, data.size() - want);
194 return p ? size_t(static_cast<const char *>(p) - data.data()) + 1 : data.size();
195 };
196
197 const size_t begin = (raw_begin == 0) ? 0 : snap(raw_begin);
198 const size_t end = snap(raw_begin + block_size);
199 if (begin >= end) { return {}; }
200 return padded_string_view(data.data() + begin, end - begin,
201 data.capacity() - begin);
202}
