570simdjson_fastfloat_strncasecmp(UC const *actual_mixedcase, UC const *expected_lowercase,

573 SIMDJSON_FASTFLOAT_IF_CONSTEXPR17(sizeof(UC) == 1) { mask = 0x2020202020202020; }
574 else SIMDJSON_FASTFLOAT_IF_CONSTEXPR17(sizeof(UC) == 2) {

585 for (size_t i = 0; i < length; i++) {
586 if ((actual_mixedcase[i] | 32) != expected_lowercase[i]) {
587 return false;

598 ::memcpy(&val2, expected_lowercase + i, sz * sizeof(UC));

639/* Helper C++14 constexpr generic implementation of leading_zeroes */
640simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR14 int
641leading_zeroes_generic(uint64_t input_num, int last_bit = 0) {
642 if (input_num & uint64_t(0xffffffff00000000)) {
643 input_num >>= 32;
644 last_bit |= 32;
645 }
646 if (input_num & uint64_t(0xffff0000)) {
647 input_num >>= 16;
648 last_bit |= 16;
649 }
650 if (input_num & uint64_t(0xff00)) {
651 input_num >>= 8;
652 last_bit |= 8;
653 }
654 if (input_num & uint64_t(0xf0)) {
655 input_num >>= 4;
656 last_bit |= 4;
657 }
658 if (input_num & uint64_t(0xc)) {
659 input_num >>= 2;
660 last_bit |= 2;
661 }
662 if (input_num & uint64_t(0x2)) { /* input_num >>= 1; */
663 last_bit |= 1;
664 }
665 return 63 - last_bit;
666}

668/* result might be undefined when input_num is zero */
669simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR20 int
670leading_zeroes(uint64_t input_num) {
671 assert(input_num > 0);
672 if (cpp20_and_in_constexpr()) {
673 return leading_zeroes_generic(input_num);
674 }
675#ifdef SIMDJSON_FASTFLOAT_VISUAL_STUDIO
676#if defined(_M_X64) || defined(_M_ARM64)
677 unsigned long leading_zero = 0;
678 // Search the mask data from most significant bit (MSB)
679 // to least significant bit (LSB) for a set bit (1).
680 _BitScanReverse64(&leading_zero, input_num);
681 return static_cast<int>(63 - leading_zero);
682#else
683 return leading_zeroes_generic(input_num);
684#endif
685#else
686 return __builtin_clzll(input_num);
687#endif
688}

690/* Helper C++14 constexpr generic implementation of countr_zero for 32-bit */
691simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR14 int
692countr_zero_generic_32(uint32_t input_num) {
693 if (input_num == 0) {
694 return 32;
695 }
696 int last_bit = 0;
697 if (!(input_num & 0x0000FFFF)) {
698 input_num >>= 16;
699 last_bit |= 16;
700 }
701 if (!(input_num & 0x00FF)) {
702 input_num >>= 8;
703 last_bit |= 8;
704 }
705 if (!(input_num & 0x0F)) {
706 input_num >>= 4;
707 last_bit |= 4;
708 }
709 if (!(input_num & 0x3)) {
710 input_num >>= 2;
711 last_bit |= 2;
712 }
713 if (!(input_num & 0x1)) {
714 last_bit |= 1;
715 }
716 return last_bit;
717}

719/* count trailing zeroes for 32-bit integers */

721countr_zero_32(uint32_t input_num) {

736// slow emulation routine for 32-bit
737simdjson_fastfloat_really_inline constexpr uint64_t emulu(uint32_t x, uint32_t y) {
738 return x * static_cast<uint64_t>(y);
739}
740
741simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR14 uint64_t
742umul128_generic(uint64_t ab, uint64_t cd, uint64_t *hi) {
743 uint64_t ad =
744 emulu(static_cast<uint32_t>(ab >> 32), static_cast<uint32_t>(cd));
745 uint64_t bd = emulu(static_cast<uint32_t>(ab), static_cast<uint32_t>(cd));
746 uint64_t adbc =
747 ad + emulu(static_cast<uint32_t>(ab), static_cast<uint32_t>(cd >> 32));
748 uint64_t adbc_carry = static_cast<uint64_t>(adbc < ad);
749 uint64_t lo = bd + (adbc << 32);
750 *hi =
751 emulu(static_cast<uint32_t>(ab >> 32), static_cast<uint32_t>(cd >> 32)) +
752 (adbc >> 32) + (adbc_carry << 32) + static_cast<uint64_t>(lo < bd);
753 return lo;
754}

770simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR20 value128
771full_multiplication(uint64_t a, uint64_t b) {
772 if (cpp20_and_in_constexpr()) {
773 value128 answer;
774 answer.low = umul128_generic(a, b, &answer.high);
775 return answer;
776 }
777 value128 answer;
778#if defined(_M_ARM64) && !defined(__MINGW32__)
779 // ARM64 has native support for 64-bit multiplications, no need to emulate
780 // But MinGW on ARM64 doesn't have native support for 64-bit multiplications
781 answer.high = __umulh(a, b);
782 answer.low = a * b;
783#elif defined(SIMDJSON_FASTFLOAT_32BIT) || (defined(_WIN64) && !defined(__clang__) && \
784 !defined(_M_ARM64) && !defined(__GNUC__))
785 answer.low = _umul128(a, b, &answer.high); // _umul128 not available on ARM64
786#elif defined(SIMDJSON_FASTFLOAT_64BIT) && defined(__SIZEOF_INT128__)
787 __uint128_t r = static_cast<__uint128_t>(a) * b;
788 answer.low = uint64_t(r);
789 answer.high = uint64_t(r >> 64);
790#else
791 answer.low = umul128_generic(a, b, &answer.high);
792#endif
793 return answer;
794}

2420 (std::is_same<T, std::uint16_t>::value && sizeof(UC) == 1)) {

2425 value = 0;
2426 answer.ec = std::errc();

2428 } else {
2429 answer.ec = std::errc::invalid_argument;
2430 answer.ptr = first;
2431 }
2432 return answer;
2433 }

2436 uint32_t digits = read4_to_u32(p);
2437 if (is_made_of_four_digits_fast(digits)) {
2438 uint32_t v = parse_four_digits_unrolled(digits);
2439 if (len >= 5 && is_integer(p[4])) {
2440 v = v * 10 + uint32_t(p[4] - '0');
2441 if (len >= 6 && is_integer(p[5])) {
2442 answer.ec = std::errc::result_out_of_range;
2443 const UC *q = p + 5;
2444 while (q != pend && is_integer(*q)) {
2445 q++;
2446 }
2447 answer.ptr = q;
2448 return answer;
2449 }
2450 if (v > 65535) {
2451 answer.ec = std::errc::result_out_of_range;
2452 answer.ptr = p + 5;
2453 return answer;
2454 }
2455 value = uint16_t(v);
2456 answer.ec = std::errc();
2457 answer.ptr = p + 5;
2458 return answer;
2459 }
2460 // 4 digits
2461 value = uint16_t(v);
2462 answer.ec = std::errc();
2463 answer.ptr = p + 4;
2464 return answer;
2465 }
2466 }
2467 }
2468 }
2469
2470 uint64_t i = 0;
2471 if (base == 10) {
2472 loop_parse_if_eight_digits(p, pend, i); // use SIMD if possible
2473 }
2474 while (p != pend) {
2475 uint8_t digit = ch_to_digit(*p);
2476 if (digit >= base) {
2477 break;
2478 }
2479 i = uint64_t(base) * i + digit; // might overflow, check this later
2480 p++;
2481 }
2482
2483 size_t digit_count = size_t(p - start_digits);
2484
2485 if (digit_count == 0) {
2486 if (has_leading_zeros) {
2487 value = 0;
2488 answer.ec = std::errc();
2489 answer.ptr = p;
2490 } else {
2491 answer.ec = std::errc::invalid_argument;
2492 answer.ptr = first;
2493 }
2494 return answer;
2495 }
2496
2497 answer.ptr = p;
2498
2499 // check u64 overflow
2500 size_t max_digits = max_digits_u64(base);
2501 if (digit_count > max_digits) {
2502 answer.ec = std::errc::result_out_of_range;
2503 return answer;
2504 }
2505 // this check can be eliminated for all other types, but they will all require
2506 // a max_digits(base) equivalent
2507 if (digit_count == max_digits) {
2508 // At the max_digits boundary the accumulator `i` may have wrapped around
2509 // 2^64. A plain `i < min_safe_u64(base)` test is not sufficient: for any
2510 // base whose max_digits-length range exceeds 2^64 (base 10 reaches
2511 // ~5.4 * 2^64 at 20 digits) the value can wrap a whole multiple of 2^64 and
2512 // land back above min_safe, slipping through. Decide exactly in O(1) using
2513 // the leading digit, following the approach used in simdjson:
2514 // ms == min_safe_u64(base) == base^(max_digits-1), the smallest
2515 // max_digits-length value.
2516 // dmax == the largest leading digit whose number can still fit in u64.
2517 // The leading-digit band [d*ms, (d+1)*ms) has width ms < 2^64, so within
2518 // the single band where d == dmax the value straddles 2^64 at most once,
2519 // and a single threshold separates wrapped from non-wrapped values. A
2520 // leading digit above dmax always overflows; below dmax always fits.
2521 uint64_t const ms = min_safe_u64(base);
2522 uint64_t const dmax = (std::numeric_limits<uint64_t>::max)() / ms;
2523 uint64_t const lead = ch_to_digit(*start_digits);
2524 if (lead > dmax || (lead == dmax && i < dmax * ms)) {
2525 answer.ec = std::errc::result_out_of_range;
2526 return answer;
2527 }
2528 }
2529
2530 // check other types overflow
2531 if (!std::is_same<T, uint64_t>::value) {
2532 if (i > uint64_t((std::numeric_limits<T>::max)()) + uint64_t(negative)) {
2533 answer.ec = std::errc::result_out_of_range;
2534 return answer;
2535 }
2536 }
2537
2538 if (negative) {
2539#ifdef SIMDJSON_FASTFLOAT_VISUAL_STUDIO
2540#pragma warning(push)
2541#pragma warning(disable : 4146)
2542#endif
2543 // this weird workaround is required because:
2544 // - converting unsigned to signed when its value is greater than signed max
2545 // is UB pre-C++23.
2546 // - reinterpret_casting (~i + 1) would work, but it is not constexpr
2547 // this is always optimized into a neg instruction (note: T is an integer
2548 // type)
2549 value = T(-(std::numeric_limits<T>::max)() -
2550 T(i - uint64_t((std::numeric_limits<T>::max)())));
2551#ifdef SIMDJSON_FASTFLOAT_VISUAL_STUDIO
2552#pragma warning(pop)
2553#endif
2554 } else {
2555 value = T(i);
2556 }
2557
2558 answer.ec = std::errc();
2559 return answer;
2560}

3287// This will compute or rather approximate w * 5**q and return a pair of 64-bit
3288// words approximating the result, with the "high" part corresponding to the
3289// most significant bits and the low part corresponding to the least significant
3290// bits.
3291//
3292template <int bit_precision>
3293simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR20 value128
3294compute_product_approximation(int64_t q, uint64_t w) {
3295 int const index = 2 * int(q - powers::smallest_power_of_five);
3296 // For small values of q, e.g., q in [0,27], the answer is always exact
3297 // because The line value128 firstproduct = full_multiplication(w,
3298 // power_of_five_128[index]); gives the exact answer.
3299 value128 firstproduct =
3300 full_multiplication(w, powers::power_of_five_128[index]);
3301 static_assert((bit_precision >= 0) && (bit_precision <= 64),
3302 " precision should be in (0,64]");
3303 constexpr uint64_t precision_mask =
3304 (bit_precision < 64) ? (uint64_t(0xFFFFFFFFFFFFFFFF) >> bit_precision)
3305 : uint64_t(0xFFFFFFFFFFFFFFFF);
3306 if ((firstproduct.high & precision_mask) ==
3307 precision_mask) { // could further guard with (lower + w < lower)
3308 // regarding the second product, we only need secondproduct.high, but our
3309 // expectation is that the compiler will optimize this extra work away if
3310 // needed.
3311 value128 secondproduct =
3312 full_multiplication(w, powers::power_of_five_128[index + 1]);
3313 firstproduct.low += secondproduct.high;
3314 if (secondproduct.high > firstproduct.low) {
3315 firstproduct.high++;
3316 }
3317 }
3318 return firstproduct;
3319}

3342// create an adjusted mantissa, biased by the invalid power2
3343// for significant digits already multiplied by 10 ** q.
3344template <typename binary>
3345simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR14 adjusted_mantissa
3346compute_error_scaled(int64_t q, uint64_t w, int lz) noexcept {
3347 int hilz = int(w >> 63) ^ 1;
3348 adjusted_mantissa answer;
3349 answer.mantissa = w << hilz;
3350 int bias = binary::mantissa_explicit_bits() - binary::minimum_exponent();
3351 answer.power2 = int32_t(detail::power(int32_t(q)) + bias - hilz - lz - 62 +
3352 invalid_am_bias);
3353 return answer;

3356// w * 10 ** q, without rounding the representation up.
3357// the power2 in the exponent will be adjusted by invalid_am_bias.
3358template <typename binary>
3359simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR20 adjusted_mantissa
3360compute_error(int64_t q, uint64_t w) noexcept {
3361 int lz = leading_zeroes(w);
3362 w <<= lz;
3363 value128 product =
3364 compute_product_approximation<binary::mantissa_explicit_bits() + 3>(q, w);
3365 return compute_error_scaled<binary>(q, product.high, lz);

3368// Computers w * 10 ** q.
3369// The returned value should be a valid number that simply needs to be
3370// packed. However, in some very rare cases, the computation will fail. In such
3371// cases, we return an adjusted_mantissa with a negative power of 2: the caller
3372// should recompute in such cases.
3373template <typename binary>
3374simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR20 adjusted_mantissa
3375compute_float(int64_t q, uint64_t w) noexcept {
3376 adjusted_mantissa answer;
3377 if ((w == 0) || (q < binary::smallest_power_of_ten())) {
3378 answer.power2 = 0;
3379 answer.mantissa = 0;
3380 // result should be zero
3381 return answer;
3382 }
3383 if (q > binary::largest_power_of_ten()) {
3384 // we want to get infinity:
3385 answer.power2 = binary::infinite_power();
3386 answer.mantissa = 0;
3387 return answer;
3388 }
3389 // At this point in time q is in [powers::smallest_power_of_five,
3390 // powers::largest_power_of_five].
3391
3392 // We want the most significant bit of i to be 1. Shift if needed.
3393 int lz = leading_zeroes(w);
3394 w <<= lz;
3395
3396 // The required precision is binary::mantissa_explicit_bits() + 3 because
3397 // 1. We need the implicit bit
3398 // 2. We need an extra bit for rounding purposes
3399 // 3. We might lose a bit due to the "upperbit" routine (result too small,
3400 // requiring a shift)
3401
3402 value128 product =
3403 compute_product_approximation<binary::mantissa_explicit_bits() + 3>(q, w);
3404 // The computed 'product' is always sufficient.
3405 // Mathematical proof:
3406 // Noble Mushtak and Daniel Lemire, Fast Number Parsing Without Fallback (to
3407 // appear) See script/mushtak_lemire.py
3408
3409 // The "compute_product_approximation" function can be slightly slower than a
3410 // branchless approach: value128 product = compute_product(q, w); but in
3411 // practice, we can win big with the compute_product_approximation if its
3412 // additional branch is easily predicted. Which is best is data specific.
3413 int upperbit = int(product.high >> 63);
3414 int shift = upperbit + 64 - binary::mantissa_explicit_bits() - 3;
3415
3416 answer.mantissa = product.high >> shift;
3417
3418 answer.power2 = int32_t(detail::power(int32_t(q)) + upperbit - lz -
3419 binary::minimum_exponent());
3420 if (answer.power2 <= 0) { // we have a subnormal?
3421 // Here have that answer.power2 <= 0 so -answer.power2 >= 0
3422 if (-answer.power2 + 1 >=
3423 64) { // if we have more than 64 bits below the minimum exponent, you
3424 // have a zero for sure.
3425 answer.power2 = 0;
3426 answer.mantissa = 0;
3427 // result should be zero
3428 return answer;
3429 }
3430 // next line is safe because -answer.power2 + 1 < 64
3431 answer.mantissa >>= -answer.power2 + 1;
3432 // Thankfully, we can't have both "round-to-even" and subnormals because
3433 // "round-to-even" only occurs for powers close to 0 in the 32-bit and
3434 // and 64-bit case (with no more than 19 digits).
3435 answer.mantissa += (answer.mantissa & 1); // round up
3436 answer.mantissa >>= 1;
3437 // There is a weird scenario where we don't have a subnormal but just.
3438 // Suppose we start with 2.2250738585072013e-308, we end up
3439 // with 0x3fffffffffffff x 2^-1023-53 which is technically subnormal
3440 // whereas 0x40000000000000 x 2^-1023-53 is normal. Now, we need to round
3441 // up 0x3fffffffffffff x 2^-1023-53 and once we do, we are no longer
3442 // subnormal, but we can only know this after rounding.
3443 // So we only declare a subnormal if we are smaller than the threshold.
3444 answer.power2 =
3445 (answer.mantissa < (uint64_t(1) << binary::mantissa_explicit_bits()))
3446 ? 0
3447 : 1;
3448 return answer;
3449 }
3450
3451 // usually, we round *up*, but if we fall right in between and and we have an
3452 // even basis, we need to round down
3453 // We are only concerned with the cases where 5**q fits in single 64-bit word.
3454 if ((product.low <= 1) && (q >= binary::min_exponent_round_to_even()) &&
3455 (q <= binary::max_exponent_round_to_even()) &&
3456 ((answer.mantissa & 3) == 1)) { // we may fall between two floats!
3457 // To be in-between two floats we need that in doing
3458 // answer.mantissa = product.high >> (upperbit + 64 -
3459 // binary::mantissa_explicit_bits() - 3);
3460 // ... we dropped out only zeroes. But if this happened, then we can go
3461 // back!!!
3462 if ((answer.mantissa << shift) == product.high) {
3463 answer.mantissa &= ~uint64_t(1); // flip it so that we do not round up
3464 }
3465 }
3466
3467 answer.mantissa += (answer.mantissa & 1); // round up
3468 answer.mantissa >>= 1;
3469 if (answer.mantissa >= (uint64_t(2) << binary::mantissa_explicit_bits())) {
3470 answer.mantissa = (uint64_t(1) << binary::mantissa_explicit_bits());
3471 answer.power2++; // undo previous addition
3472 }
3473
3474 answer.mantissa &= ~(uint64_t(1) << binary::mantissa_explicit_bits());
3475 if (answer.power2 >= binary::infinite_power()) { // infinity
3476 answer.power2 = binary::infinite_power();
3477 answer.mantissa = 0;
3478 }
3479 return answer;
3480}

3694// add two small integers, checking for overflow.
3695// we want an efficient operation. for msvc, where
3696// we don't have built-in intrinsics, this is still
3697// pretty fast.

3699scalar_add(limb x, limb y, bool &overflow) noexcept {

3701// gcc and clang

3703#if __has_builtin(__builtin_add_overflow)

3706 return z;

3717// multiply two small integers, getting both the high and low bits.
3718simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR20 limb
3719scalar_mul(limb x, limb y, limb &carry) noexcept {
3720#ifdef SIMDJSON_FASTFLOAT_64BIT_LIMB
3721#if defined(__SIZEOF_INT128__)
3722 // GCC and clang both define it as an extension.
3723 __uint128_t z = __uint128_t(x) * __uint128_t(y) + __uint128_t(carry);
3724 carry = limb(z >> limb_bits);
3725 return limb(z);
3726#else
3727 // fallback, no native 128-bit integer multiplication with carry.
3728 // on msvc, this optimizes identically, somehow.
3729 value128 z = full_multiplication(x, y);
3730 bool overflow;
3731 z.low = scalar_add(z.low, carry, overflow);
3732 z.high += uint64_t(overflow); // cannot overflow
3733 carry = z.high;
3734 return z.low;
3735#endif
3736#else
3737 uint64_t z = uint64_t(x) * uint64_t(y) + uint64_t(carry);
3738 carry = limb(z >> limb_bits);
3739 return limb(z);
3740#endif
3741}

3783// add bigint to bigint starting from index.
3784// used in grade school multiplication
3785template <uint16_t size>
3786SIMDJSON_FASTFLOAT_CONSTEXPR20 bool large_add_from(stackvec<size> &x, limb_span y,
3787 size_t start) noexcept {
3788 // the effective x buffer is from `xstart..x.len()`, so exit early
3789 // if we can't get that current range.
3790 if (x.len() < start || y.len() > x.len() - start) {
3791 SIMDJSON_FASTFLOAT_TRY(x.try_resize(y.len() + start, 0));
3792 }
3793
3794 bool carry = false;
3795 for (size_t index = 0; index < y.len(); index++) {
3796 limb xi = x[index + start];
3797 limb yi = y[index];
3798 bool c1 = false;
3799 bool c2 = false;
3800 xi = scalar_add(xi, yi, c1);
3801 if (carry) {
3802 xi = scalar_add(xi, 1, c2);
3803 }
3804 x[index + start] = xi;
3805 carry = c1 | c2;
3806 }
3807
3808 // handle overflow
3809 if (carry) {
3810 SIMDJSON_FASTFLOAT_TRY(small_add_from(x, 1, y.len() + start));
3811 }
3812 return true;

3994 // returns true if we were able to shift all the digits.
3995 SIMDJSON_FASTFLOAT_CONSTEXPR20 bool shl_bits(size_t n) noexcept {
3996 // Internally, for each item, we shift left by n, and add the previous
3997 // right shifted limb-bits.
3998 // For example, we transform (for u8) shifted left 2, to:
3999 // b10100100 b01000010
4000 // b10 b10010001 b00001000
4001 SIMDJSON_FASTFLOAT_DEBUG_ASSERT(n != 0);
4002 SIMDJSON_FASTFLOAT_DEBUG_ASSERT(n < sizeof(limb) * 8);
4003
4004 size_t shl = n;
4005 size_t shr = limb_bits - shl;
4006 limb prev = 0;
4007 for (size_t index = 0; index < vec.len(); index++) {
4008 limb xi = vec[index];
4009 vec[index] = (xi << shl) | (prev >> shr);
4010 prev = xi;
4011 }
4012
4013 limb carry = prev >> shr;
4014 if (carry != 0) {
4015 return vec.try_push(carry);
4016 }
4017 return true;
4018 }

4157// this algorithm is not even close to optimized, but it has no practical
4158// effect on performance: in order to have a faster algorithm, we'd need
4159// to slow down performance for faster algorithms, and this is still fast.
4160simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR14 int32_t
4161scientific_exponent(uint64_t mantissa, int32_t exponent) noexcept {
4162 while (mantissa >= 10000) {
4163 mantissa /= 10000;
4164 exponent += 4;
4165 }
4166 while (mantissa >= 100) {
4167 mantissa /= 100;
4168 exponent += 2;
4169 }
4170 while (mantissa >= 10) {
4171 mantissa /= 10;
4172 exponent += 1;
4173 }
4174 return exponent;

4177// this converts a native floating-point number to an extended-precision float.
4178template <typename T>
4179simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR20 adjusted_mantissa
4180to_extended(T value) noexcept {
4181 using equiv_uint = equiv_uint_t<T>;
4182 constexpr equiv_uint exponent_mask = binary_format<T>::exponent_mask();
4183 constexpr equiv_uint mantissa_mask = binary_format<T>::mantissa_mask();
4184 constexpr equiv_uint hidden_bit_mask = binary_format<T>::hidden_bit_mask();
4185
4186 adjusted_mantissa am;
4187 int32_t bias = binary_format<T>::mantissa_explicit_bits() -
4188 binary_format<T>::minimum_exponent();
4189 equiv_uint bits;
4190#if SIMDJSON_FASTFLOAT_HAS_BIT_CAST
4191 bits = std::bit_cast<equiv_uint>(value);
4192#else
4193 ::memcpy(&bits, &value, sizeof(T));
4194#endif
4195 if ((bits & exponent_mask) == 0) {
4196 // denormal
4197 am.power2 = 1 - bias;
4198 am.mantissa = bits & mantissa_mask;
4199 } else {
4200 // normal
4201 am.power2 = int32_t((bits & exponent_mask) >>
4202 binary_format<T>::mantissa_explicit_bits());
4203 am.power2 -= bias;
4204 am.mantissa = (bits & mantissa_mask) | hidden_bit_mask;
4205 }
4206
4207 return am;
4208}

4210// get the extended precision value of the halfway point between b and b+u.
4211// we are given a native float that represents b, so we need to adjust it
4212// halfway between b and b+u.
4213template <typename T>
4214simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR20 adjusted_mantissa
4215to_extended_halfway(T value) noexcept {
4216 adjusted_mantissa am = to_extended(value);
4217 am.mantissa <<= 1;
4218 am.mantissa += 1;
4219 am.power2 -= 1;
4220 return am;
4221}

4223// round an extended-precision float to the nearest machine float.
4224template <typename T, typename callback>
4225simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR14 void round(adjusted_mantissa &am,
4226 callback cb) noexcept {
4227 int32_t mantissa_shift = 64 - binary_format<T>::mantissa_explicit_bits() - 1;
4228 if (-am.power2 >= mantissa_shift) {
4229 // have a denormal float
4230 int32_t shift = -am.power2 + 1;
4231 cb(am, (shift < 64 ? shift : 64));
4232 // check for round-up: if rounding-nearest carried us to the hidden bit.
4233 am.power2 = (am.mantissa <
4234 (uint64_t(1) << binary_format<T>::mantissa_explicit_bits()))
4235 ? 0
4236 : 1;
4237 return;
4238 }
4239
4240 // have a normal float, use the default shift.
4241 cb(am, mantissa_shift);
4242
4243 // check for carry
4244 if (am.mantissa >=
4245 (uint64_t(2) << binary_format<T>::mantissa_explicit_bits())) {
4246 am.mantissa = (uint64_t(1) << binary_format<T>::mantissa_explicit_bits());
4247 am.power2++;
4248 }
4249
4250 // check for infinite: we could have carried to an infinite power
4251 am.mantissa &= ~(uint64_t(1) << binary_format<T>::mantissa_explicit_bits());
4252 if (am.power2 >= binary_format<T>::infinite_power()) {
4253 am.power2 = binary_format<T>::infinite_power();
4254 am.mantissa = 0;
4255 }
4256}

4258template <typename callback>
4259simdjson_fastfloat_really_inline SIMDJSON_FASTFLOAT_CONSTEXPR14 void
4260round_nearest_tie_even(adjusted_mantissa &am, int32_t shift,
4261 callback cb) noexcept {
4262 uint64_t const mask = (shift == 64) ? UINT64_MAX : (uint64_t(1) << shift) - 1;
4263 uint64_t const halfway = (shift == 0) ? 0 : uint64_t(1) << (shift - 1);
4264 uint64_t truncated_bits = am.mantissa & mask;
4265 bool is_above = truncated_bits > halfway;
4266 bool is_halfway = truncated_bits == halfway;
4267
4268 // shift digits into position
4269 if (shift == 64) {
4270 am.mantissa = 0;
4271 } else {
4272 am.mantissa >>= shift;
4273 }
4274 am.power2 += shift;
4275
4276 bool is_odd = (am.mantissa & 1) == 1;
4277 am.mantissa += uint64_t(cb(is_odd, is_halfway, is_above));
4278}

4374// parse the significant digits into a big integer
4375template <typename UC>
4376inline SIMDJSON_FASTFLOAT_CONSTEXPR20 void
4377parse_mantissa(bigint &result, parsed_number_string_t<UC> &num,
4378 size_t max_digits, size_t &digits) noexcept {
4379 // try to minimize the number of big integer and scalar multiplication.
4380 // therefore, try to parse 8 digits at a time, and multiply by the largest
4381 // scalar value (9 or 19 digits) for each step.
4382 size_t counter = 0;
4383 digits = 0;
4384 limb value = 0;
4385#ifdef SIMDJSON_FASTFLOAT_64BIT_LIMB
4386 size_t step = 19;
4387#else
4388 size_t step = 9;
4389#endif
4390
4391 // process all integer digits.
4392 UC const *p = num.integer.ptr;
4393 UC const *pend = p + num.integer.len();
4394 skip_zeros(p, pend);
4395 // process all digits, in increments of step per loop
4396 while (p != pend) {
4397 while ((std::distance(p, pend) >= 8) && (step - counter >= 8) &&
4398 (max_digits - digits >= 8)) {
4399 parse_eight_digits(p, value, counter, digits);
4400 }
4401 while (counter < step && p != pend && digits < max_digits) {
4402 parse_one_digit(p, value, counter, digits);
4403 }
4404 if (digits == max_digits) {
4405 // add the temporary value, then check if we've truncated any digits
4406 add_native(result, limb(powers_of_ten_uint64[counter]), value);
4407 bool truncated = is_truncated(p, pend);
4408 if (num.fraction.ptr != nullptr) {
4409 truncated |= is_truncated(num.fraction);
4410 }
4411 if (truncated) {
4412 round_up_bigint(result, digits);
4413 }
4414 return;
4415 } else {
4416 add_native(result, limb(powers_of_ten_uint64[counter]), value);
4417 counter = 0;
4418 value = 0;
4419 }
4420 }
4421
4422 // add our fraction digits, if they're available.
4423 if (num.fraction.ptr != nullptr) {
4424 p = num.fraction.ptr;
4425 pend = p + num.fraction.len();
4426 if (digits == 0) {
4427 skip_zeros(p, pend);
4428 }
4429 // process all digits, in increments of step per loop
4430 while (p != pend) {
4431 while ((std::distance(p, pend) >= 8) && (step - counter >= 8) &&
4432 (max_digits - digits >= 8)) {
4433 parse_eight_digits(p, value, counter, digits);
4434 }
4435 while (counter < step && p != pend && digits < max_digits) {
4436 parse_one_digit(p, value, counter, digits);
4437 }
4438 if (digits == max_digits) {
4439 // add the temporary value, then check if we've truncated any digits
4440 add_native(result, limb(powers_of_ten_uint64[counter]), value);
4441 bool truncated = is_truncated(p, pend);
4442 if (truncated) {
4443 round_up_bigint(result, digits);
4444 }
4445 return;
4446 } else {
4447 add_native(result, limb(powers_of_ten_uint64[counter]), value);
4448 counter = 0;
4449 value = 0;
4450 }
4451 }
4452 }
4453
4454 if (counter != 0) {
4455 add_native(result, limb(powers_of_ten_uint64[counter]), value);
4456 }
4457}

4459template <typename T>
4460inline SIMDJSON_FASTFLOAT_CONSTEXPR20 adjusted_mantissa
4461positive_digit_comp(bigint &bigmant, int32_t exponent) noexcept {
4462 SIMDJSON_FASTFLOAT_ASSERT(bigmant.pow10(uint32_t(exponent)));
4463 adjusted_mantissa answer;
4464 bool truncated;
4465 answer.mantissa = bigmant.hi64(truncated);
4466 int bias = binary_format<T>::mantissa_explicit_bits() -
4467 binary_format<T>::minimum_exponent();
4468 answer.power2 = bigmant.bit_length() - 64 + bias;
4469
4470 round<T>(answer, [truncated](adjusted_mantissa &a, int32_t shift) {
4471 round_nearest_tie_even(
4472 a, shift,
4473 [truncated](bool is_odd, bool is_halfway, bool is_above) -> bool {
4474 return is_above || (is_halfway && truncated) ||
4475 (is_odd && is_halfway);
4476 });
4477 });
4478
4479 return answer;
4480}

4482// the scaling here is quite simple: we have, for the real digits `m * 10^e`,
4483// and for the theoretical digits `n * 2^f`. Since `e` is always negative,
4484// to scale them identically, we do `n * 2^f * 5^-f`, so we now have `m * 2^e`.
4485// we then need to scale by `2^(f- e)`, and then the two significant digits
4486// are of the same magnitude.
4487template <typename T>
4488inline SIMDJSON_FASTFLOAT_CONSTEXPR20 adjusted_mantissa negative_digit_comp(
4489 bigint &bigmant, adjusted_mantissa am, int32_t exponent) noexcept {
4490 bigint &real_digits = bigmant;
4491 int32_t real_exp = exponent;
4492
4493 // get the value of `b`, rounded down, and get a bigint representation of b+h
4494 adjusted_mantissa am_b = am;
4495 // gcc7 buf: use a lambda to remove the noexcept qualifier bug with
4496 // -Wnoexcept-type.
4497 round<T>(am_b,
4498 [](adjusted_mantissa &a, int32_t shift) { round_down(a, shift); });
4499 T b;
4500 to_float(false, am_b, b);
4501 adjusted_mantissa theor = to_extended_halfway(b);
4502 bigint theor_digits(theor.mantissa);
4503 int32_t theor_exp = theor.power2;
4504
4505 // scale real digits and theor digits to be same power.
4506 int32_t pow2_exp = theor_exp - real_exp;
4507 uint32_t pow5_exp = uint32_t(-real_exp);
4508 if (pow5_exp != 0) {
4509 SIMDJSON_FASTFLOAT_ASSERT(theor_digits.pow5(pow5_exp));
4510 }
4511 if (pow2_exp > 0) {
4512 SIMDJSON_FASTFLOAT_ASSERT(theor_digits.pow2(uint32_t(pow2_exp)));
4513 } else if (pow2_exp < 0) {
4514 SIMDJSON_FASTFLOAT_ASSERT(real_digits.pow2(uint32_t(-pow2_exp)));
4515 }
4516
4517 // compare digits, and use it to direct rounding
4518 int ord = real_digits.compare(theor_digits);
4519 adjusted_mantissa answer = am;
4520 round<T>(answer, [ord](adjusted_mantissa &a, int32_t shift) {
4521 round_nearest_tie_even(
4522 a, shift, [ord](bool is_odd, bool _, bool __) -> bool {
4523 static_cast<void>(_); // not needed, since we've done our comparison
4524 static_cast<void>(__); // not needed, since we've done our comparison
4525 if (ord > 0) {
4526 return true;
4527 } else if (ord < 0) {
4528 return false;
4529 } else {
4530 return is_odd;
4531 }
4532 });
4533 });
4534
4535 return answer;
4536}

4538// parse the significant digits as a big integer to unambiguously round
4539// the significant digits. here, we are trying to determine how to round
4540// an extended float representation close to `b+h`, halfway between `b`
4541// (the float rounded-down) and `b+u`, the next positive float. this
4542// algorithm is always correct, and uses one of two approaches. when
4543// the exponent is positive relative to the significant digits (such as
4544// 1234), we create a big-integer representation, get the high 64-bits,
4545// determine if any lower bits are truncated, and use that to direct
4546// rounding. in case of a negative exponent relative to the significant
4547// digits (such as 1.2345), we create a theoretical representation of
4548// `b` as a big-integer type, scaled to the same binary exponent as
4549// the actual digits. we then compare the big integer representations
4550// of both, and use that to direct rounding.
4551template <typename T, typename UC>
4552inline SIMDJSON_FASTFLOAT_CONSTEXPR20 adjusted_mantissa
4553digit_comp(parsed_number_string_t<UC> &num, adjusted_mantissa am) noexcept {
4554 // remove the invalid exponent bias
4555 am.power2 -= invalid_am_bias;
4556
4557 int32_t sci_exp =
4558 scientific_exponent(num.mantissa, static_cast<int32_t>(num.exponent));
4559 size_t max_digits = binary_format<T>::max_digits();
4560 size_t digits = 0;
4561 bigint bigmant;
4562 parse_mantissa(bigmant, num, max_digits, digits);
4563 // can't underflow, since digits is at most max_digits.
4564 int32_t exponent = sci_exp + 1 - int32_t(digits);
4565 if (exponent >= 0) {
4566 return positive_digit_comp<T>(bigmant, exponent);
4567 } else {
4568 return negative_digit_comp<T>(bigmant, am, exponent);
4569 }
4570}

4655 // A fast function to check your floating-point rounding mode
4656 // https://lemire.me/blog/2022/11/16/a-fast-function-to-check-your-floating-point-rounding-mode/
4657 //
4658 // This function is meant to be equivalent to :
4659 // prior: #include <cfenv>
4660 // return fegetround() == FE_TONEAREST;
4661 // However, it is expected to be much faster than the fegetround()
4662 // function call.

4665 // at compile-time.
4666 // There might be other ways to prevent compile-time optimizations (e.g.,

4675// fmin + 1.0f == 1.0f - fmin.
4676//
4677// FE_UPWARD:
4678// fmin + 1.0f > 1
4679// 1.0f - fmin == 1
4680//
4681// FE_DOWNWARD or FE_TOWARDZERO:
4682// fmin + 1.0f == 1
4683// 1.0f - fmin < 1

4686// enabled, as rounding conventions may not apply.

4688#pragma warning(push)

4690// see
4691// https://stackoverflow.com/questions/46079446/is-there-a-warning-for-floating-point-equality-checking-in-visual-studio-2013

4694#pragma clang diagnostic ignored "-Wfloat-equal"
4695#elif defined(__GNUC__)

4697#pragma GCC diagnostic ignored "-Wfloat-equal"

4700#ifdef SIMDJSON_FASTFLOAT_VISUAL_STUDIO
4701#pragma warning(pop)
4702#elif defined(__clang__)

4767 // we want round-to-nearest in all cases, irrespective of the rounding mode
4768 // selected on the thread.
4769 // We proceed optimistically, assuming that detail::rounds_to_nearest()
4770 // returns true.
4771 if (binary_format<T>::min_exponent_fast_path() <= exponent &&
4772 exponent <= binary_format<T>::max_exponent_fast_path() &&
4773 mantissa <= binary_format<T>::max_mantissa_fast_path()) {
4774 // The mantissa bound above is a necessary condition for BOTH branches
4775 // below: the rounding-mode-dependent branch checks the tighter
4776 // max_mantissa_fast_path(exponent) <= max_mantissa_fast_path(). Testing
4777 // it before detail::rounds_to_nearest() spares long-mantissa inputs
4778 // (which can never take the fast path) the volatile-float probe.
4779 //
4780 // Unfortunately, the conventional Clinger's fast path is only possible
4781 // when the system rounds to the nearest float.
4782 //
4783 // We expect the next branch to almost always be selected.
4784 // We could check it first (before the previous branch), but
4785 // there might be performance advantages at having the check
4786 // be last.
4787 if (!cpp20_and_in_constexpr() && detail::rounds_to_nearest()) {
4788 // We have that fegetround() == FE_TONEAREST.
4789 // Next is Clinger's fast path.
4790 value = T(mantissa);
4791 if (exponent < 0) {
4792 value = value / binary_format<T>::exact_power_of_ten(-exponent);
4793 } else {
4794 value = value * binary_format<T>::exact_power_of_ten(exponent);
4795 }
4796 if (is_negative) {
4797 value = -value;
4798 }
4799 return true;
4800 } else {
4801 // We do not have that fegetround() == FE_TONEAREST.
4802 // Next is a modified Clinger's fast path, inspired by Jakub Jelinek's
4803 // proposal
4804 if (exponent >= 0 &&
4805 mantissa <= binary_format<T>::max_mantissa_fast_path(exponent)) {
4806#if defined(__clang__) || defined(SIMDJSON_FASTFLOAT_32BIT)
4807 // Clang may map 0 to -0.0 when fegetround() == FE_DOWNWARD
4808 if (mantissa == 0) {
4809 value = is_negative ? T(-0.) : T(0.);
4810 return true;
4811 }
4812#endif
4813 value = T(mantissa) * binary_format<T>::exact_power_of_ten(exponent);
4814 if (is_negative) {
4815 value = -value;
4816 }
4817 return true;
4818 }
4819 }
4820 }
4821 return false;
4822}
