40error_code tag_invoke(deserialize_tag, auto &val, T &out) noexcept {

50 SIMDJSON_TRY(val.get_float().get(x));

52 return SUCCESS;

55 double x;
56 SIMDJSON_TRY(val.get_double().get(x));
57 out = static_cast<T>(x);
58 return SUCCESS;

90// any string-like type (can be constructed from std::string_view)
91template <concepts::constructible_from_string_view T, typename ValT>
92error_code tag_invoke(deserialize_tag, ValT &val, T &out) noexcept(std::is_nothrow_constructible_v<T, std::string_view>) {
93 std::string_view str;
94 SIMDJSON_TRY(val.get_string().get(str));
95 if constexpr (requires { out.assign(str.data(), str.size()); }) {
96 // Copy straight into out (e.g., std::string): building a temporary and
97 // move-assigning it is markedly slower.
98 out.assign(str.data(), str.size());
99 } else {
100 out = T{str};
101 }
102 return SUCCESS;
103}

124// Whether to deserialize the elements of the container T directly into a new
125// element (emplace_one(out) and then get) rather than into a temporary that is
126// then moved into the container. A temporary of a trivially copyable type lives
127// in registers and is cheap to move, and value-initializing a small element in
128// place costs more than it saves (GCC zeroes it with rep stos). But moving a
129// large element, or a string (whose deserialization then needs a temporary
130// string and a move assignment of its own), is expensive.
131template <typename T>
132concept deserialize_in_place =
133 concepts::returns_reference<T> && requires(T &c) { c.pop_back(); } &&
134 !std::is_trivially_copyable_v<typename T::value_type> &&
135 (sizeof(typename T::value_type) > 32 || concepts::constructible_from_string_view<typename T::value_type>);
136
137// Removes the last element of the container on destruction while armed.
138template <typename T>
139struct pop_back_guard {
140 T &container;
141 bool armed{true};
142 ~pop_back_guard() {
143 if (armed) { container.pop_back(); }
144 }
145};
146} // namespace details

155template <concepts::appendable_containers T, typename ValT>
156error_code tag_invoke(deserialize_tag, ValT &val, T &out) noexcept(false) {
157 using value_type = typename std::remove_cvref_t<T>::value_type;
158 static_assert(
159 deserializable<value_type, ValT>,
160 "The specified type inside the container must itself be deserializable");
161 static_assert(
162 std::is_default_constructible_v<value_type>,
163 "The specified type inside the container must default constructible.");
164 SIMDJSON_IMPLEMENTATION::ondemand::array arr;
165 if constexpr (std::is_same_v<std::remove_cvref_t<ValT>, SIMDJSON_IMPLEMENTATION::ondemand::array>) {
166 arr = val;
167 } else {
168 SIMDJSON_TRY(val.get_array().get(arr));
169 }
170
171 if constexpr (std::is_same_v<T, std::vector<value_type>> && !std::is_same_v<value_type, bool>) {
172 // Collect the elements in a per-thread scratch vector that keeps its
173 // capacity from call to call, then move them into out after reserving the
174 // exact size: out is allocated once instead of being regrown. A nested
175 // array of the same type finds the scratch busy and takes the paths below.
176 struct scratch_space {
177 std::vector<value_type> elements{};
178 bool busy{false};
179 };
180 static thread_local scratch_space scratch;
181 if (!scratch.busy && out.empty()) {
182 struct release_scratch {
183 scratch_space &s;
184 T &out;
185 size_t parsed{0};
186 bool complete{false};
187 // On an error or an exception, out gets the elements parsed so far (as
188 // with the loops below), without allocating. Kept out of the hot path.
189 simdjson_never_inline void keep_parsed() noexcept {
190 s.elements.resize(parsed);
191 out.swap(s.elements);
192 }
193 ~release_scratch() {
194 if (simdjson_unlikely(!complete)) { keep_parsed(); }
195 s.elements.clear();
196 // Do not hold on to the memory of a very large array.
197 if (s.elements.capacity() * sizeof(value_type) > (1 << 20)) { std::vector<value_type>().swap(s.elements); }
198 s.busy = false;
199 }
200 } release{scratch, out};
201 scratch.busy = true;
202 for (auto v : arr) {
203 SIMDJSON_TRY(v.get<value_type>(scratch.elements.emplace_back()));
204 release.parsed++;
205 }
206 out.reserve(release.parsed);
207 release.complete = true;
208 for (auto &e : scratch.elements) { out.emplace_back(std::move(e)); }
209 return SUCCESS;
210 }
211 }
212 if constexpr (details::deserialize_in_place<T>) {
213 for (auto v : arr) {
214 auto &slot = concepts::emplace_one(out);
215 // An error or an exception (a user tag_invoke may throw) must not leave
216 // a partially deserialized element behind.
217 details::pop_back_guard<T> guard{out};
218 SIMDJSON_TRY(v.get<value_type>(slot));
219 guard.armed = false;
220 }
221 } else {
222 for (auto v : arr) {
223 // Deserialize into a temporary first: an error or an exception (a user
224 // tag_invoke may throw) must not leave a default-constructed element behind.
225 value_type temp;
226 if (auto const err = v.get<value_type>(temp); err) {
227 return err;
228 }
229 concepts::emplace_one(out, std::move(temp));
230 }
231 }
232 return SUCCESS;
233}

317template <concepts::smart_pointer T, typename ValT>
318error_code tag_invoke(deserialize_tag, ValT &val, T &out) noexcept(false) {
319 using element_type = typename std::remove_cvref_t<T>::element_type;
320
321 // For better error messages, don't use these as constraints on
322 // the tag_invoke CPO.
323 static_assert(
324 deserializable<element_type, ValT>,
325 "The specified type inside the unique_ptr must itself be deserializable");
326 static_assert(
327 std::is_default_constructible_v<element_type>,
328 "The specified type inside the unique_ptr must default constructible.");
329
330 // Own the allocation before get(): a user tag_invoke may throw.
331 std::unique_ptr<element_type> ptr(new (std::nothrow) element_type());
332 if (!ptr) {
333 return MEMALLOC;
334 }
335 SIMDJSON_TRY(val.template get<element_type>(*ptr));
336 out = std::move(ptr);
337 return SUCCESS;
338}

343template <concepts::optional_type T>
344error_code tag_invoke(deserialize_tag, auto &val, T &out) noexcept(nothrow_deserializable<typename std::remove_cvref_t<T>::value_type, decltype(val)>) {
345 using value_type = typename std::remove_cvref_t<T>::value_type;
346
347 // Check if the value is null
348 bool is_null_value;
349 SIMDJSON_TRY( val.is_null().get(is_null_value) );
350 if (is_null_value) {
351 out.reset(); // Set to nullopt
352 return SUCCESS;
353 }
354
355 if (!out) {
356 out.emplace();
357 }
358 SIMDJSON_TRY(val.template get<value_type>(out.value()));
359 return SUCCESS;
360}

388// static reflection). It provides the compile-time machinery for building a
389// key_selector from a struct's members, the per-member helpers shared by every

397// A member participates if it is public, non-const, and not annotated with skip
398// or skip_deserializing.
399consteval bool is_eligible_member(std::meta::info mem) {
400 return !std::meta::is_const(mem) && std::meta::is_public(mem)
401 && !simdjson::detail::has_annotation(mem, ^^simdjson::detail::skip_tag)
402 && !simdjson::detail::has_annotation(mem, ^^simdjson::detail::skip_deserializing_tag);
403}

405// A field is a data member reached from T through a chain of members: usually a
406// direct member of T (a path of length one), or a member of a member annotated
407// with flatten (the members of a flattened structure are fields of the enclosing
408// one). A field is identified by the type member_path<m1, m2, ..., leaf>.
409template <std::meta::info First, std::meta::info... Rest>
410struct member_path {
411 // The data member holding the value; its annotations drive (de)serialization.
412 static constexpr std::meta::info leaf = [] {
413 std::meta::info members[] = {First, Rest...};
414 return members[sizeof...(Rest)];
415 }();
416 template <typename T>
417 static simdjson_inline constexpr auto &get(T &obj) noexcept {
418 if constexpr (sizeof...(Rest) == 0) {
419 return obj.[:First:];
420 } else {
421 return member_path<Rest...>::get(obj.[:First:]);
422 }
423 }
424};

426// True when T can be flattened: a structure deserialized member by member (not
427// a string, a container, an optional, a smart pointer, ...).
428template <typename T>
429constexpr bool flattenable_type = user_defined_type<T> && !concepts::string_view_keyed_map<T>
430 && !concepts::container_but_not_string<T> && !concepts::smart_pointer<T>;
431
432consteval void append_eligible_fields(std::meta::info type, std::vector<std::meta::info> &prefix,
433 std::vector<std::meta::info> &fields) {
434 for (std::meta::info mem : std::meta::nonstatic_data_members_of(type, std::meta::access_context::unchecked())) {
435 if (!is_eligible_member(mem)) { continue; }
436 prefix.push_back(std::meta::reflect_constant(mem));
437 if (simdjson::detail::has_annotation(mem, ^^simdjson::detail::flatten_tag)) {
438 std::meta::info flattened = simdjson::detail::flattened_type(mem);
439 if (!std::meta::extract<bool>(std::meta::substitute(^^flattenable_type, {flattened}))) {
440 throw std::meta::exception(u8"simdjson::flatten requires a member whose type is a structure deserialized member by member", mem);
441 }
442 append_eligible_fields(flattened, prefix, fields);
443 } else {
444 fields.push_back(std::meta::substitute(^^member_path, prefix));
445 }
446 prefix.pop_back();
447 }
448}

450// The fields of `type` that participate in deserialization, in declaration order
451// (the fields of a flattened member take its place). A field's position in this
452// list is its "field index" below.
453consteval std::vector<std::meta::info> eligible_fields(std::meta::info type) {
454 std::vector<std::meta::info> prefix;
455 std::vector<std::meta::info> fields;
456 append_eligible_fields(type, prefix, fields);
457 return fields;
458}

465// Number of fields that participate in deserialization. A class can have zero

474// The keys accepted for a member (or an enumerator): its JSON key followed by its
475// aliases. They are static strings, so they can be used as template arguments.
476consteval std::vector<const char *> accepted_keys_of(std::meta::info entity) {
477 std::vector<const char *> keys;
478 for (std::string_view key : simdjson::detail::json_key_names(entity)) {
479 bool repeated = false;
480 for (const char *previous : keys) { repeated = repeated || std::string_view(previous) == key; }
481 if (!repeated) { keys.push_back(std::define_static_string(key)); }
482 }
483 return keys;
484}

486// Every key accepted for T: the eligible fields in order, each followed by its
487// aliases. This is the key list of T's key_selector.
488consteval std::vector<const char *> accepted_keys(std::meta::info type) {
489 std::vector<const char *> keys;
490 for (std::meta::info path : eligible_fields(type)) {
491 for (const char *key : accepted_keys_of(field_leaf(path))) { keys.push_back(key); }
492 }
493 return keys;
494}

496// The field index of each entry of accepted_keys(type).
497consteval std::vector<std::size_t> accepted_key_fields(std::meta::info type) {
498 std::vector<std::size_t> key_fields;
499 std::vector<std::meta::info> fields = eligible_fields(type);
500 for (std::size_t i = 0; i < fields.size(); ++i) {
501 for (std::size_t j = 0; j < accepted_keys_of(field_leaf(fields[i])).size(); ++j) { key_fields.push_back(i); }
502 }
503 return key_fields;
504}

506// True when no two eligible fields of T accept the same key. Otherwise one JSON
507// key would have to fill several members: this is reported at compile time.
508template <typename T>
509consteval bool accepted_keys_are_distinct() {
510 std::vector<const char *> keys = accepted_keys(^^T);
511 for (std::size_t i = 0; i < keys.size(); ++i) {
512 for (std::size_t j = i + 1; j < keys.size(); ++j) {
513 if (std::string_view(keys[i]) == std::string_view(keys[j])) { return false; }
514 }
515 }
516 return true;
517}

519// True when some accepted key of T is written with escape sequences in JSON (a
520// double quote, a backslash or a control character). Such a key can only be
521// matched by comparing unescaped keys (deserialize_struct_scan): obj[key] and
522// the key_selector compare the raw bytes.
523template <typename T>
524consteval bool keys_need_unescaping() {
525 for (std::string_view key : accepted_keys(^^T)) {
526 for (char c : key) {
527 if (c == '\\' || c == '"' || static_cast<unsigned char>(c) < 0x20) { return true; }
528 }
529 }
530 return false;

595// True when T's accepted keys satisfy every key_selector requirement, so a
596// key_selector can be built for T without a compile-time error. This mirrors the
597// key_selector limits (see key_selector.h): at most 255 keys, each key non-empty
598// and at most 63 characters, no backslash / double-quote / null byte, and all
599// keys distinct. Keys with any other control character are excluded too: they
600// are escaped in JSON, so their raw bytes never match. When this returns false
601// the deserializer falls back to deserialize_struct_scan instead of failing to
602// compile.
603template <typename T>
604consteval bool keys_fit_selector() {
605 std::vector<std::string_view> keys;
606 for (const char *key : accepted_keys(^^T)) { keys.push_back(key); }
607 if (keys.size() > 255) { return false; }
608 for (std::size_t i = 0; i < keys.size(); ++i) {
609 if (keys[i].empty() || keys[i].size() > 63) { return false; }
610 for (char c : keys[i]) {
611 if (c == '\\' || c == '"' || static_cast<unsigned char>(c) < 0x20) { return false; }
612 }
613 for (std::size_t j = i + 1; j < keys.size(); ++j) {
614 if (keys[i] == keys[j]) { return false; }
615 }
616 }
617 return true;
618}

628// Deserialize a JSON value into `target`, the storage of member `mem`, through
629// the member's with<Adapter> annotation when the adapter provides a deserialize
630// function.
631template <auto mem, typename ValueT, typename M>
632simdjson_warn_unused simdjson_inline error_code deserialize_member_value(ValueT &field_value, M &target) noexcept(false) {
633 constexpr std::meta::info with_type = simdjson::detail::annotation_of_template(mem, ^^simdjson::detail::with_t);
634 if constexpr (with_type != std::meta::info{}) {
635 using adapter = typename [: with_type :]::adapter;
636 using ondemand_value = SIMDJSON_IMPLEMENTATION::ondemand::value;
637 if constexpr (requires { { adapter::deserialize(field_value, target) } -> std::convertible_to<error_code>; }) {
638 return adapter::deserialize(field_value, target);
639 } else if constexpr (requires(ondemand_value &v) { { adapter::deserialize(v, target) } -> std::convertible_to<error_code>; }
640 && requires { field_value.get_value(); }) {
641 // A transparent structure read from a document: the adapter takes an
642 // ondemand::value. A scalar document cannot be viewed as a value, so it
643 // reports SCALAR_DOCUMENT_AS_VALUE (an adapter taking auto& receives the
644 // document itself and has no such limitation).
645 ondemand_value v;
646 SIMDJSON_TRY(field_value.get_value().get(v));
647 return adapter::deserialize(v, target);
648 } else {
649 static_assert(!declares_deserialize(^^adapter),
650 "the deserialize function of a simdjson::with adapter must be callable as "
651 "Adapter::deserialize(simdjson::ondemand::value &, T &) and return an error_code");
652 return field_value.get(target);
653 }
654 } else {
655 return field_value.get(target);
656 }
657}

659// Deserialize the storage `target` of member `mem` from a JSON value.
660template <auto mem, typename ValueT, typename M>
661simdjson_warn_unused simdjson_inline error_code deserialize_member(ValueT &field_value, M &target) noexcept(false) {
662 if constexpr (has_default<mem>() && std::is_default_constructible_v<M> && std::is_move_assignable_v<M>) {
663 // A present key replaces the default value: deserialize into a fresh
664 // temporary so that, e.g., a container does not append to its default
665 // content, and a failure leaves the default untouched.
666 M value{};
667 SIMDJSON_TRY(deserialize_member_value<mem>(field_value, value));
668 target = std::move(value);
669 return SUCCESS;
670 } else {
671 return deserialize_member_value<mem>(field_value, target);
672 }
673}

675// Deserialize the field with the given field index.
676template <typename T>
677simdjson_warn_unused simdjson_inline error_code deserialize_field_at(
678 std::size_t field_index, SIMDJSON_IMPLEMENTATION::ondemand::value field_value, T &out) noexcept(false) {
679 std::size_t counter = 0;
680 template for (constexpr auto path : std::define_static_array(eligible_fields(^^T))) {
681 using field = [: path :];
682 if (field_index == counter) { return deserialize_member<field::leaf>(field_value, field::get(out)); }
683 ++counter;
684 }
685 return SUCCESS;
686}

688// Called when the key(s) of member `mem`, stored in `target`, are missing from
689// the JSON object: an error for a required member, a call to the default_from
690// factory, or nothing at all.
691template <auto mem, typename M>
692simdjson_warn_unused simdjson_inline error_code on_missing_member(M &target) noexcept(false) {
693 constexpr std::meta::info default_from_type = simdjson::detail::annotation_of_template(mem, ^^simdjson::detail::default_from_t);
694 if constexpr (default_from_type != std::meta::info{}) {
695 target = [: default_from_type :]::factory();
696 return SUCCESS;
697 } else if constexpr (may_be_absent<mem>()) {
698 // For optional and default_value members, a missing key is not an error:
699 // leave the member at its current (default) value.
700 (void)target;
701 return SUCCESS;
702 } else {
703 (void)target;
704 return NO_SUCH_FIELD;
705 }
706}

708// Report or handle every field that was not seen in the JSON object.
709template <typename T, std::size_t N>
710simdjson_warn_unused simdjson_inline error_code handle_missing_fields(
711 const std::array<bool, N> &seen_field, T &out) noexcept(false) {
712 std::size_t counter = 0;
713 template for (constexpr auto path : std::define_static_array(eligible_fields(^^T))) {
714 using field = [: path :];
715 if (!seen_field[counter]) { SIMDJSON_TRY(on_missing_member<field::leaf>(field::get(out))); }
716 ++counter;
717 }
718 return SUCCESS;
719}

721// Ordered, per-field deserialization: one obj[key] lookup per eligible field
722// (and per alias, until one is found). This is the opt-out path
723// (-DSIMDJSON_DISABLE_KEY_SELECTOR_REFLECTION=1), except for structs with keys
724// that need unescaping (see keys_need_unescaping).
725template <typename T>
726simdjson_warn_unused error_code deserialize_struct_ordered(
727 SIMDJSON_IMPLEMENTATION::ondemand::object &obj, T &out) noexcept(false) {
728 template for (constexpr auto path : std::define_static_array(eligible_fields(^^T))) {
729 using field = [: path :];
730 SIMDJSON_IMPLEMENTATION::ondemand::value field_value;
731 error_code error = NO_SUCH_FIELD;
732 template for (constexpr const char *key : std::define_static_array(accepted_keys_of(field::leaf))) {
733 if (error == NO_SUCH_FIELD) { error = obj[std::string_view(key)].get(field_value); }
734 }
735 if (error == NO_SUCH_FIELD) {
736 SIMDJSON_TRY(on_missing_member<field::leaf>(field::get(out)));
737 } else if (error) {
738 return error;
739 } else {
740 SIMDJSON_TRY(deserialize_member<field::leaf>(field_value, field::get(out)));
741 }
742 }
743 return SUCCESS;
744}

746// Appends the JSON keys that serialization writes for members of `type` that
747// deserialization cannot assign (const or non-public members, directly or
748// through flatten). `all` is true inside a flattened member that is itself
749// unassignable. Keys of skip_deserializing members are not included: they are
750// unknown keys, as documented.
751consteval void append_unassignable_keys(std::meta::info type, bool all, std::vector<const char *> &keys) {
752 for (std::meta::info mem : std::meta::nonstatic_data_members_of(type, std::meta::access_context::unchecked())) {
753 if (simdjson::detail::has_annotation(mem, ^^simdjson::detail::skip_tag)
754 || simdjson::detail::has_annotation(mem, ^^simdjson::detail::skip_serializing_tag)
755 || simdjson::detail::has_annotation(mem, ^^simdjson::detail::skip_deserializing_tag)) {
756 continue;
757 }
758 bool unassignable = all || !is_eligible_member(mem);
759 if (simdjson::detail::has_annotation(mem, ^^simdjson::detail::flatten_tag)) {
760 append_unassignable_keys(simdjson::detail::flattened_type(mem), unassignable, keys);
761 } else if (unassignable) {
762 keys.push_back(std::define_static_string(simdjson::detail::json_key_name(mem)));
763 }
764 }
765}

767consteval std::vector<const char *> unassignable_keys(std::meta::info type) {
768 std::vector<const char *> keys;
769 append_unassignable_keys(type, false, keys);
770 return keys;
771}

773// Deserialization by a single pass over every field of the object, comparing
774// unescaped keys. It is used for structs annotated with deny_unknown_fields
775// (DenyUnknown = true), where a key that does not map to an eligible field is
776// reported as UNKNOWN_FIELD, and as the fallback for structs whose keys the
777// key_selector or obj[key] cannot match (see keys_fit_selector and
778// keys_need_unescaping). As with object::for_each, the first occurrence of a
779// field wins (later duplicates, or aliases of a field already seen, are ignored).
780template <bool DenyUnknown, typename T>
781simdjson_warn_unused error_code deserialize_struct_scan(
782 SIMDJSON_IMPLEMENTATION::ondemand::object &obj, T &out) noexcept(false) {
783 static constexpr auto keys = std::define_static_array(accepted_keys(^^T));
784 static constexpr auto key_fields = std::define_static_array(accepted_key_fields(^^T));
785 std::array<bool, eligible_field_count<T>()> seen_field{};
786 for (auto field_result : obj) {
787 SIMDJSON_IMPLEMENTATION::ondemand::field json_field;
788 SIMDJSON_TRY(std::move(field_result).get(json_field));
789 std::string_view key;
790 SIMDJSON_TRY(json_field.unescaped_key().get(key));
791 std::size_t key_index = keys.size();
792 for (std::size_t i = 0; i < keys.size(); ++i) {
793 if (key == std::string_view(keys[i])) { key_index = i; break; }
794 }
795 if (key_index == keys.size()) {
796 if constexpr (DenyUnknown) {
797 // A key that T itself serializes (e.g. of a const member) is not
798 // unknown: a serialized value must parse back.
799 static constexpr auto ignored_keys = std::define_static_array(unassignable_keys(^^T));
800 bool ignored = false;
801 for (const char *ignored_key : ignored_keys) {
802 if (key == std::string_view(ignored_key)) { ignored = true; break; }
803 }
804 if (!ignored) { return UNKNOWN_FIELD; }
805 }
806 continue;
807 }
808 const std::size_t field_index = key_fields[key_index];
809 if (seen_field[field_index]) { continue; }
810 seen_field[field_index] = true;
811 SIMDJSON_TRY(deserialize_field_at(field_index, json_field.value(), out));
812 }
813 return handle_missing_fields(seen_field, out);
814}

823// Deserialize a reflected struct. By default this builds a compile-time
824// key_selector from the struct's members and walks each object once with
825// object::for_each (perfect-hash key matching), instead of one obj[key] lookup
826// per member. Other paths are used instead:
827// - globally, the ordered per-member path when defining
828// -DSIMDJSON_DISABLE_KEY_SELECTOR_REFLECTION=1;
829// - automatically and per-type, a scan of the object comparing unescaped keys
830// when the struct's keys do not fit the key_selector limits (see
831// keys_fit_selector) or cannot be compared raw (see keys_need_unescaping),
832// so that long member names and the like keep compiling rather than
833// tripping a static_assert.
834// Structs annotated with deny_unknown_fields always use the strict scan, and
835// structs annotated with transparent are deserialized as their single member.
836//
837// noexcept(false): a member's tag_invoke may throw and the exception must reach
838// the caller of get<T>().
839template <typename T, typename ValT>
840 requires(user_defined_type<T> && std::is_class_v<T>)
841error_code tag_invoke(deserialize_tag, ValT &val, T &out) noexcept(false) {
842 if constexpr (key_selector_reflection_detail::is_transparent<T>()) {
843 constexpr auto mem = simdjson::detail::transparent_member(^^T);
844 if constexpr (std::is_same_v<std::remove_cvref_t<ValT>, SIMDJSON_IMPLEMENTATION::ondemand::object>) {
845 // We were handed an object: only a structure can be deserialized from it.
846 if constexpr (user_defined_type<typename [: std::meta::type_of(mem) :]>) {
847 return tag_invoke(deserialize_tag{}, val, out.[:mem:]);
848 } else {
849 return INCORRECT_TYPE;
850 }
851 } else {
852 return key_selector_reflection_detail::deserialize_member<mem>(val, out.[:mem:]);
853 }
854 } else {
855 static_assert(key_selector_reflection_detail::accepted_keys_are_distinct<T>(),
856 "two members of this structure accept the same JSON key (check rename, alias, "
857 "rename_all and flatten)");
858 SIMDJSON_IMPLEMENTATION::ondemand::object obj;
859 if constexpr (std::is_same_v<std::remove_cvref_t<ValT>, SIMDJSON_IMPLEMENTATION::ondemand::object>) {
860 obj = val;
861 } else {
862 SIMDJSON_TRY(val.get_object().get(obj));
863 }
864 if constexpr (simdjson::detail::has_annotation(^^T, ^^simdjson::detail::deny_unknown_fields_tag)) {
865 return key_selector_reflection_detail::deserialize_struct_scan<true>(obj, out);
866 } else {
867#if defined(SIMDJSON_DISABLE_KEY_SELECTOR_REFLECTION) && SIMDJSON_DISABLE_KEY_SELECTOR_REFLECTION
868 // Opt-out build: use the ordered per-member path, unless obj[key] cannot
869 // match T's keys.
870 if constexpr (key_selector_reflection_detail::keys_need_unescaping<T>()) {
871 return key_selector_reflection_detail::deserialize_struct_scan<false>(obj, out);
872 } else {
873 return key_selector_reflection_detail::deserialize_struct_ordered(obj, out);
874 }
875#else
876 if constexpr (key_selector_reflection_detail::eligible_field_count<T>() == 0) {
877 // No fields to deserialize: an empty key_selector cannot be built, so just
878 // validate that the input is an object (done above) and succeed. Mirrors the
879 // ordered per-member path, which iterates over zero members.
880 (void)out;
881 (void)obj;
882 return SUCCESS;
883 } else if constexpr (!key_selector_reflection_detail::keys_fit_selector<T>()) {
884 // Automatic fallback: T's accepted keys do not fit the key_selector limits
885 // (e.g. a member name longer than 63 characters, or a key with a double
886 // quote), so building a selector would be a compile error. Scan the object
887 // instead, so the default never breaks a struct that the opt-out path would
888 // accept.
889 return key_selector_reflection_detail::deserialize_struct_scan<false>(obj, out);
890 } else {
891 using selector = key_selector_reflection_detail::selector_for<T>;
892 if constexpr (key_selector_reflection_detail::all_eligible_fields_required<T>()
893 && !key_selector_reflection_detail::has_aliases<T>()) {
894 // Fast path: every member is required and has a single key. A single
895 // for_each pass parses each matched field; the returned match count then
896 // tells us whether every member was present (matched_count ==
897 // selector::size()) without a per-member "seen" array. A value-parse error
898 // (e.g. a type mismatch) is propagated by for_each.
899 auto walk = obj.template for_each<selector>(
900 [&](std::size_t matched_index, SIMDJSON_IMPLEMENTATION::ondemand::value field_value) -> error_code {
901 std::size_t counter = 0;
902 template for (constexpr auto path : std::define_static_array(key_selector_reflection_detail::eligible_fields(^^T))) {
903 using field = [: path :];
904 if (matched_index == counter) { return key_selector_reflection_detail::deserialize_member<field::leaf>(field_value, field::get(out)); }
905 ++counter;
906 }
907 return SUCCESS;
908 });
909 if (walk.error) { return walk.error; }
910 // A missing required member shows up as a short match count and is reported as
911 // NO_SUCH_FIELD, mirroring the ordered obj[key] path.
912 if (walk.matched_count != selector::size()) { return NO_SUCH_FIELD; }
913 return SUCCESS;
914 } else {
915 static constexpr auto key_fields = std::define_static_array(key_selector_reflection_detail::accepted_key_fields(^^T));
916 std::array<bool, key_selector_reflection_detail::eligible_field_count<T>()> seen_field{};
917 // Single pass over the object: each field whose key matches a member (or one
918 // of its aliases) yields its selector index, which we map back to the
919 // corresponding member. The first key seen for a member wins. The callback
920 // returns an error_code so that a value-parse error (e.g. a type mismatch on
921 // a matched field) is propagated by for_each instead of being silently dropped.
922 error_code walk_error = obj.template for_each<selector>(
923 [&](std::size_t matched_index, SIMDJSON_IMPLEMENTATION::ondemand::value field_value) -> error_code {
924 const std::size_t field_index = key_fields[matched_index];
925 if (seen_field[field_index]) { return SUCCESS; }
926 seen_field[field_index] = true;
927 return key_selector_reflection_detail::deserialize_field_at(field_index, field_value, out);
928 });
929 if (walk_error) { return walk_error; }
930 // Required members must be present: a missing one is reported as
931 // NO_SUCH_FIELD, mirroring the ordered obj[key] path. Optional and defaulted
932 // members may be absent.
933 return key_selector_reflection_detail::handle_missing_fields(seen_field, out);
934 }
935 }
936#endif // SIMDJSON_DISABLE_KEY_SELECTOR_REFLECTION
937 }
938 }
939}

941// Support for enum deserialization - deserialize from string representation using
942// expand approach from P2996R12. The accepted strings are the enumerator's JSON key
943// (see rename and rename_all) and its aliases.
944template <typename T, typename ValT>
945 requires(std::is_enum_v<T>)
946error_code tag_invoke(deserialize_tag, ValT &val, T &out) noexcept {
947#if SIMDJSON_STATIC_REFLECTION
948 std::string_view str;
949 SIMDJSON_TRY(val.get_string().get(str));
950 static constexpr auto enumerators = std::define_static_array(std::meta::enumerators_of(^^T));
951 template for (constexpr auto enum_val : enumerators) {
952 template for (constexpr const char *key : std::define_static_array(key_selector_reflection_detail::accepted_keys_of(enum_val))) {
953 if (str == std::string_view(key)) {
954 out = [:enum_val:];
955 return SUCCESS;
956 }
957 }
958 };
959
960 return INCORRECT_TYPE;
961#else
962 // Fallback: deserialize as integer if reflection not available
963 std::underlying_type_t<T> int_val;
964 SIMDJSON_TRY(val.get(int_val));
965 out = static_cast<T>(int_val);
966 return SUCCESS;
967#endif
968}
